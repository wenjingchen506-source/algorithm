def longestConsecutive(self, nums: List[int]) -> int:
    s = set(nums)
    res = 0
    for x in nums:
        if x not in s: continue
        cur = 1
        s.remove(x)
        y = x + 1
        while y in s:
            s.remove(y)
            cur, y = cur + 1, y + 1
        y = x - 1
        while y in s:
            s.remove(y)
            cur, y = cur + 1, y - 1
        res = max(res, cur)
    return res

class Solution:
    def findOriginalArray(self, changed: List[int]) -> List[int]:
        n = len(changed)
        if n & 1: return []
        cnt=Counter(changed)
        ans=[]
        for i ,x in enumerate(changed):
            if cnt[x]==0:continue#漏了，核心就是处理所有cnt，循环动力
            if x==0:
                if cnt[x]&1 :return []

                    ans.extend([0]*(cnt[0]//2))
                cnt[x]=0

                    y=4*y
        return ans

def bisect_left(lo, hi, check_left):
    while lo < hi:
        mid = (lo + hi) // 2
        if check_left(mid):
            hi = mid       # 如果是 True，说明当前点可能是第一个，或者第一个在左边，收缩右边界
        else:
            lo = mid + 1   # 如果是 False，说明太小了，第一个 True 必然在右边，步进左边界

class Solution:
    def minSessions(self, tasks: List[int], sessionTime: int) -> int:
        tasks.sort(reverse=True)
        n=len(tasks)
        cnt=[0]*n
        res=float('inf')
        def dfs(i,cur):
            nonlocal res
            if cur>res:return 
            if i==n:
                res=cur
                return

                if tasks[i]<=sessionTime-cnt[j]:
                    cnt[j]+=tasks[i]

                    cnt[j]-=tasks[i]
            cnt[cur]+=tasks[i]
            dfs(i+1,cur+1)
            cnt[cur]-=tasks[i]
        dfs(0,0)
        return res

把路径用 join 存成字符串的例子

回想一下我们之前聊过的“连线”灾难： 如果你写的是 out.append(arr)， 那你就是往 out 里塞了一根指向 arr 这个木箱子的红线（内存引用）。 后续 arr.pop() 会把结果全毁掉。 但是！请死死盯住你写的 join 操作到底干了什么： str(i) for i in arr：首先，它把篮子里的数字（比如 1 和 2），变成了字符 '1' 和 '2'。 ' '.join(...)：然后，它用空格把这两个字符拼接起来， 在内存中凭空创造了一个全新的字符串对象："1 2"。 out.append("1 2")：最后，你把这个崭新的字符串，放进了 out 数组里。 真相大白： 在这段代码里，你根本没有把 arr 这个列表塞进结果集里！你是在每一次触底的时候，利用 join 自动把 arr 里的内容“冲洗成了一张固定的实物照片（字符串）”， 然后把照片放进了相册 out 里！

        unique_effects = set()
        for r in range(min(n, m)):

            lamp2 = (2 * r) % m

import sys

def solve():
    data = sys.stdin.read().split()
    if not data: return
    n = int(data[0])
    s = [int(data[i]) for i in range(1, len(data), 2)]
    b = [int(data[i]) for i in range(2, len(data), 2)]
    
    ans = float('inf')
    
    # 1 << n 相当于 2 的 n 次方。例如 n=3，范围是 1 到 7 (即二进制 001 到 111)
    # 从 1 开始，天然排除了 000 (什么都不选) 的非法情况
    for mask in range(1, 1 << n):
        cur_s = 1
        cur_b = 0
        
        # 检查这个整数的每一位
        for i in range(n):
            # 位运算：(1 << i) 是制作一个掩码，& 操作判断第 i 位是否为 1
            if mask & (1 << i):
                cur_s *= s[i]
                cur_b += b[i]
                
        ans = min(ans, abs(cur_s - cur_b))
        
    print(ans)

solve()

欧拉筛与最小质因子表草稿。第 237 行判定要核对 spf 和 isp

##法四：欧式筛(此版本即可马上筛查一个数是不是素数、如果是合数其最小质因子、所有素数)
##在外层枚举数字 i 的同时，内层枚举已知的质数 p，划掉它们的乘积 i * p；
##一旦发现 i 已经是 p 的倍数，立刻停止内层枚举，从而保证所有合数只被其最小质因子处理。    
    def eulerwithspf(n):
        spf=list(range(n+1))
        isprime=[True]*(n+1)
        isprime[0]=isprime[1]=False
        prime=[]
        for i in range(2,n+1):
            
            if isp[i]==i:
                # 即if is_prime[i]:如果 i没被划掉，那它一定是质数，收编进prime数组
                prime.append(i)
                
            for p in prime:
                if i*p>n:# 越界保护：如果乘积已经超过了我们要找的范围 n，直接跳出
                    break
                
                # 划掉合数：用最小质因子 p 乘以 i，把它们的乘积划掉
                spf[i*p]=p
                isprime[i*p]=False
                
            # 【欧拉筛的灵魂核心！！！】
            # 如果 i 已经被当前的质数 p 整除，立刻打断循环！
            # 绝对不能再用后面的质数去乘这个 i 了！
                if i%p==0:
                    break
        return prime,spf# 返回收集到的所有质数,以及所有数最大质因子
